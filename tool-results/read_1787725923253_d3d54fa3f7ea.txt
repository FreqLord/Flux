    16→/* ── Tone → progress-bar fill class / color ── */
    17→function toneFill(tone: string): { className?: string; color?: string } {
    18→  switch (tone) {
    19→    case "acc":   return { className: "pf pf-acc" };
    20→    case "red":   return { className: "pf pf-red" };
    21→    case "amb":   return { className: "pf pf-amb" };
    22→    case "teal":  return { className: "pf pf-teal" };
    23→    case "grn":   return { className: "pf pf-grn" };
    24→    case "indigo": return { color: "var(--indigo)" };
    25→    case "t2":    return { color: "var(--t3)" };
    26→    default:      return { className: "pf pf-acc" };
    27→  }
    28→}
    29→
    30→function statusFor(spent: number, limit: number): { cls: string; label: string } {
    31→  if (limit <= 0) return { cls: "badge bk", label: "—" };
    32→  const r = spent / limit;
    33→  if (r > 1) return { cls: "badge br", label: "Over" };
    34→  if (r >= 0.9) return { cls: "badge ba", label: "Tight" };
    35→  return { cls: "badge bg", label: "Under" };
    36→}
    37→
    38→/* ── Transaction helpers ── */
    39→function txCategoryBadge(category: string): string {
    40→  switch (category) {
    41→    case "Income":    return "badge bg";
    42→    case "Food":      return "badge br";
    43→    case "Utilities": return "badge ba";
    44→    case "Vault":     return "badge bt";
    45→    case "Tools":     return "badge ba";
    46→    case "Transport": return "badge br";
    47→    default:          return "badge bk";
    48→  }
    49→}

    51→function txAmountColor(flow: string): string {
    52→  switch (flow) {
    53→    case "in":    return "var(--grn)";
    54→    case "out":   return "var(--red)";
    55→    case "vault": return "var(--teal)";
    56→    default:      return "var(--t1)";
    57→  }
    58→}

    60→function txAmountSign(flow: string): string {
    61→  switch (flow) {
    62→    case "in":  return "+";
    63→    case "out": return "−";
    64→    default:    return "";
    65→  }
    66→}

    68→function formatTxDate(date: string | Date): string {
    69→  const d = typeof date === "string" ? new Date(date) : date;
    70→  if (isNaN(d.getTime())) return "—";
    71→  const months = ["Jan", "Feb", "Mar", "Apr", "May", "Jun", "Jul", "Aug", "Sep", "Oct", "Nov", "Dec"];
    72→  return `${months[d.getMonth()]} ${d.getDate()}`;
    73→}

   664→/* ─── Month-over-month mini comparison card ─── */
   665→function MonthOverMonth({
   666→  thisMonth,
   667→  lastMonth,
   668→  thisLabel,
   669→  lastLabel,
   670→}: {
   671→  thisMonth: number;
   672→  lastMonth: number;
   673→  thisLabel: string;
   674→  lastLabel: string;
   675→}) {
   676→  const delta = thisMonth - lastMonth;
   677→  const pctDelta = lastMonth > 0 ? (delta / lastMonth) * 100 : 0;
   678→  const isFlat = delta === 0;
   679→  const isDown = delta < 0;
   680→  const maxBar = Math.max(thisMonth, lastMonth, 1);
   681→  const hThis = Math.max(10, Math.round((thisMonth / maxBar) * 96));
   682→  const hLast = Math.max(10, Math.round((lastMonth / maxBar) * 96));
   683→  const deltaColor = isFlat ? "var(--t3)" : isDown ? "var(--grn)" : "var(--red)";
   684→  const deltaBg = isFlat ? "var(--bg3)" : isDown ? "var(--grnd)" : "var(--redd)";
   685→
   686→  return (
   687→    <div className="card card-sm">
   688→      <div className="card-t" style={{ marginBottom: 4 }}>Month-over-month</div>
   689→      <div className="card-s" style={{ marginBottom: 12 }}>Spending vs last month</div>
   690→
   691→      <div style={{ display: "flex", alignItems: "baseline", gap: 8, marginBottom: 4 }}>
   692→        <span
   693→          className="flux-mono"
   694→          style={{
   695→            fontSize: 22,
   696→            fontWeight: 600,
   697→            color: "var(--t1)",
   698→            letterSpacing: "-.02em",
   699→          }}
   700→        >
   701→          {formatINR(thisMonth, { compact: true })}
   702→        </span>
   703→        <span
   704→          className="flux-mono"
   705→          style={{
   706→            display: "inline-flex",
   707→            alignItems: "center",
   708→            gap: 3,
   709→            fontSize: 11,
   710→            fontWeight: 700,
   711→            color: deltaColor,
   712→            background: deltaBg,
   713→            padding: "2px 7px",
   714→            borderRadius: 100,
   715→          }}
   716→        >
   717→          <Icon name={isFlat ? "info" : isDown ? "down" : "up"} size={11} />
   718→          {isFlat ? "0%" : `${Math.abs(Math.round(pctDelta))}%`}
   719→        </span>
   720→      </div>
   721→      <div style={{ fontSize: 11, color: "var(--t3)", marginBottom: 14 }}>
   722→        vs {formatINR(lastMonth, { compact: true })} in {lastLabel}
   723→      </div>
   724→
   725→      {/* 2-bar mini comparison */}
   726→      <div
   727→        style={{
   728→          display: "flex",
   729→          alignItems: "end",
   730→          justifyContent: "center",
   731→          gap: 24,
   732→          height: 110,
   733→        }}
   734→      >
   735→        <div
   736→          style={{
   737→            display: "flex",
   738→            flexDirection: "column",
   739→            alignItems: "center",
   740→            gap: 6,
   741→            flex: "0 0 56px",
   742→          }}
   743→        >
   744→          <div
   745→            className="flux-mono"
   746→            style={{ fontSize: 10.5, fontWeight: 600, color: "var(--t2)" }}
   747→          >
   748→            {formatINR(lastMonth, { compact: true })}
   749→          </div>
   750→          <div
   751→            style={{
   752→              width: "100%",
   753→              height: hLast,
   754→              background: "var(--bg3)",
   755→              borderRadius: "6px 6px 0 0",
   756→            }}
   757→          />
   758→          <div className="label-sm" style={{ marginBottom: 0 }}>{lastLabel}</div>
   759→        </div>
   760→        <div
   761→          style={{
   762→            display: "flex",
   763→            flexDirection: "column",
   764→            alignItems: "center",
   765→            gap: 6,
   766→            flex: "0 0 56px",
   767→          }}
   768→        >
   769→          <div
   770→            className="flux-mono"
   771→            style={{ fontSize: 10.5, fontWeight: 600, color: "var(--amb)" }}
   772→          >
   773→            {formatINR(thisMonth, { compact: true })}
   774→          </div>
   775→          <div
   776→            style={{
   777→              width: "100%",
   778→              height: hThis,
   779→              background: "var(--amb)",
   780→              borderRadius: "6px 6px 0 0",
   781→            }}
   782→          />
   783→          <div className="label-sm" style={{ marginBottom: 0 }}>{thisLabel}</div>
   784→        </div>
   785→      </div>
   786→    </div>
   787→  );
   788→}
