    11→/** "Arjun Kumar" → "AK" */
    12→function initials(name: string): string {
    13→  if (!name) return "FX";
    14→  const parts = name.trim().split(/\s+/);
    15→  if (parts.length === 1) return parts[0].slice(0, 2).toUpperCase();
    16→  return (parts[0][0] + parts[parts.length - 1][0]).toUpperCase();
    17→}

    39→function formatGoal(key: GoalKey, p: FluxProfile): string {
    40→  if (key === "minRunwayMonths") {
    41→    return `${p.minRunwayMonths.toFixed(1)} mo`;
    42→  }
    43→  return formatINR(p[key] as number);
    44→}
