    24→/* ── Tone → progress-bar fill class / color ── */
    25→function toneFill(tone: string): { className?: string; color?: string } {
    26→  switch (tone) {
    27→    case "acc":   return { className: "pf pf-acc" };
    28→    case "red":   return { className: "pf pf-red" };
    29→    case "amb":   return { className: "pf pf-amb" };
    30→    case "teal":  return { className: "pf pf-teal" };
    31→    case "grn":   return { className: "pf pf-grn" };
    32→    case "indigo": return { color: "var(--indigo)" };
    33→    case "t2":    return { color: "var(--t3)" };
    34→    default:      return { className: "pf pf-acc" };
    35→  }
    36→}
    37→
    38→function statusFor(spent: number, limit: number): { cls: string; label: string } {
    39→  if (limit <= 0) return { cls: "badge bk", label: "—" };
    40→  const r = spent / limit;
    41→  if (r > 1) return { cls: "badge br", label: "Over" };
    42→  if (r >= 0.9) return { cls: "badge ba", label: "Tight" };
    43→  return { cls: "badge bg", label: "Under" };
    44→}

    46→/* ── Transaction helpers ── */
    47→function txCategoryBadge(category: string): string {
    48→  switch (category) {
    49→    case "Income":    return "badge bg";
    50→    case "Food":      return "badge br";
    51→    case "Utilities": return "badge ba";
    52→    case "Vault":     return "badge bt";
    53→    case "Tools":     return "badge ba";
    54→    case "Transport": return "badge br";
    55→    default:          return "badge bk";
    56→  }
    57→}

    59→function txAmountColor(flow: string): string {
    60→  switch (flow) {
    61→    case "in":    return "var(--grn)";

    64→    default:      return "var(--t1)";

    76→function formatTxDate(date: string | Date): string {
    77→  const d = typeof date === "string" ? new Date(date) : date;
    78→  if (isNaN(d.getTime())) return "—";
    79→  const months = ["Jan", "Feb", "Mar", "Apr", "May", "Jun", "Jul", "Aug", "Sep", "Oct", "Nov", "Dec"];
    80→  return `${months[d.getMonth()]} ${d.getDate()}`;
    81→}

   783→/* ─── Month-over-month mini comparison card ─── */
   784→function MonthOverMonth({
   785→  thisMonth,
   786→  lastMonth,
   787→  thisLabel,
   788→  lastLabel,
   789→}: {
   790→  thisMonth: number;
   791→  lastMonth: number;
   792→  thisLabel: string;
   793→  lastLabel: string;
   794→}) {
   795→  const delta = thisMonth - lastMonth;
   796→  const pctDelta = lastMonth > 0 ? (delta / lastMonth) * 100 : 0;
   797→  const isFlat = delta === 0;
   798→  const isDown = delta < 0;
   799→  const maxBar = Math.max(thisMonth, lastMonth, 1);
   800→  const hThis = Math.max(10, Math.round((thisMonth / maxBar) * 96));
   801→  const hLast = Math.max(10, Math.round((lastMonth / maxBar) * 96));
   802→  const deltaColor = isFlat ? "var(--t3)" : isDown ? "var(--grn)" : "var(--red)";
   803→  const deltaBg = isFlat ? "var(--bg3)" : isDown ? "var(--grnd)" : "var(--redd)";
   804→
   805→  return (
   806→    <div className="card card-sm">
   807→      <div className="card-t" style={{ marginBottom: 4 }}>Month-over-month</div>
   808→      <div className="card-s" style={{ marginBottom: 12 }}>Spending vs last month</div>
   809→
   810→      <div style={{ display: "flex", alignItems: "baseline", gap: 8, marginBottom: 4 }}>
   811→        <span
   812→          className="flux-mono"
   813→          style={{
   814→            fontSize: 22,
   815→            fontWeight: 600,
   816→            color: "var(--t1)",
   817→            letterSpacing: "-.02em",
   818→          }}
   819→        >
   820→          {formatINR(thisMonth, { compact: true })}
   821→        </span>
   822→        <span
   823→          className="flux-mono"
   824→          style={{
   825→            display: "inline-flex",
   826→            alignItems: "center",
   827→            gap: 3,
   828→            fontSize: 11,
   829→            fontWeight: 700,
   830→            color: deltaColor,
   831→            background: deltaBg,
   832→            padding: "2px 7px",
   833→            borderRadius: 100,
   834→          }}
   835→        >
   836→          <Icon name={isFlat ? "info" : isDown ? "down" : "up"} size={11} />
   837→          {isFlat ? "0%" : `${Math.abs(Math.round(pctDelta))}%`}
   838→        </span>
   839→      </div>
   840→      <div style={{ fontSize: 11, color: "var(--t3)", marginBottom: 14 }}>
   841→        vs {formatINR(lastMonth, { compact: true })} in {lastLabel}
   842→      </div>
   843→
   844→      {/* 2-bar mini comparison */}
   845→      <div
   846→        style={{
   847→          display: "flex",
   848→          alignItems: "end",
   849→          justifyContent: "center",
   850→          gap: 24,
   851→          height: 110,
   852→        }}
   853→      >
   854→        <div
   855→          style={{
   856→            display: "flex",
   857→            flexDirection: "column",
   858→            alignItems: "center",
   859→            gap: 6,
   860→            flex: "0 0 56px",
   861→          }}
   862→        >
   863→          <div
   864→            className="flux-mono"
   865→            style={{ fontSize: 10.5, fontWeight: 600, color: "var(--t2)" }}
   866→          >
   867→            {formatINR(lastMonth, { compact: true })}
   868→          </div>
   869→          <div
   870→            style={{
   871→              width: "100%",
   872→              height: hLast,
   873→              background: "var(--bg3)",
   874→              borderRadius: "6px 6px 0 0",
   875→            }}
   876→          />
   877→          <div className="label-sm" style={{ marginBottom: 0 }}>{lastLabel}</div>
   878→        </div>
   879→        <div
   880→          style={{
   881→            display: "flex",
   882→            flexDirection: "column",
   883→            alignItems: "center",
   884→            gap: 6,
   885→            flex: "0 0 56px",
   886→          }}
   887→        >
   888→          <div
   889→            className="flux-mono"
   890→            style={{ fontSize: 10.5, fontWeight: 600, color: "var(--amb)" }}
   891→          >
   892→            {formatINR(thisMonth, { compact: true })}
   893→          </div>
   894→          <div
   895→            style={{
   896→              width: "100%",
   897→              height: hThis,
   898→              background: "var(--amb)",
   899→              borderRadius: "6px 6px 0 0",
   900→            }}
   901→          />
   902→          <div className="label-sm" style={{ marginBottom: 0 }}>{thisLabel}</div>
   903→        </div>
   904→      </div>
   905→    </div>
   906→  );
