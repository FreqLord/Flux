    17→    17→function toneFill(tone: string): { className?: string; color?: string } {
    18→    18→  switch (tone) {
    19→    19→    case "acc":   return { className: "pf pf-acc" };
    20→    20→    case "red":   return { className: "pf pf-red" };
    21→    21→    case "amb":   return { className: "pf pf-amb" };
    22→    22→    case "teal":  return { className: "pf pf-teal" };
    23→    23→    case "grn":   return { className: "pf pf-grn" };
    24→    24→    case "indigo": return { color: "var(--indigo)" };
    25→    25→    case "t2":    return { color: "var(--t3)" };
    26→    26→    default:      return { className: "pf pf-acc" };
    27→    27→  }
    28→    28→}

    30→    30→function statusFor(spent: number, limit: number): { cls: string; label: string } {
    31→    31→  if (limit <= 0) return { cls: "badge bk", label: "—" };
    32→    32→  const r = spent / limit;
    33→    33→  if (r > 1) return { cls: "badge br", label: "Over" };
    34→    34→  if (r >= 0.9) return { cls: "badge ba", label: "Tight" };
    35→    35→  return { cls: "badge bg", label: "Under" };
    36→    36→}

    68→    68→function formatTxDate(date: string | Date): string {
    69→    69→  const d = typeof date === "string" ? new Date(date) : date;
    70→    70→  if (isNaN(d.getTime())) return "—";
    71→    71→  const months = ["Jan", "Feb", "Mar", "Apr", "May", "Jun", "Jul", "Aug", "Sep", "Oct", "Nov", "Dec"];
    72→    72→  return `${months[d.getMonth()]} ${d.getDate()}`;
    73→    73→}

   665→   665→function MonthOverMonth({
   666→   666→  thisMonth,
   667→   667→  lastMonth,
   668→   668→  thisLabel,

   674→   674→  lastLabel: string;

   677→   677→  const pctDelta = lastMonth > 0 ? (delta / lastMonth) * 100 : 0;
   678→   678→  const isFlat = delta === 0;
   679→   679→  const isDown = delta < 0;

   681→   681→  const hThis = Math.max(10, Math.round((thisMonth / maxBar) * 96));
   682→   682→  const hLast = Math.max(10, Math.round((lastMonth / maxBar) * 96));
   683→   683→  const deltaColor = isFlat ? "var(--t3)" : isDown ? "var(--grn)" : "var(--red)";
   684→   684→  const deltaBg = isFlat ? "var(--bg3)" : isDown ? "var(--grnd)" : "var(--redd)";
